cmake_minimum_required(VERSION 3.22.1)
project(fgallery_tiff)
include(FetchContent)
# Match the headers used by AndroidLibRaw 2.0.7; only headers are compiled here.
FetchContent_Declare(libraw_headers URL https://codeload.github.com/LibRaw/LibRaw/tar.gz/9646d776c7c61976080a8f2be67928df0750493e)
FetchContent_GetProperties(libraw_headers)
if(NOT libraw_headers_POPULATED)
    FetchContent_Populate(libraw_headers)
endif()
add_library(fgallery_tiff SHARED tiff_decode.cpp)
target_include_directories(fgallery_tiff PRIVATE ${libraw_headers_SOURCE_DIR})
target_compile_features(fgallery_tiff PRIVATE cxx_std_17)
target_link_libraries(fgallery_tiff dl jnigraphics)
target_link_options(fgallery_tiff PRIVATE "-Wl,-z,max-page-size=16384")
